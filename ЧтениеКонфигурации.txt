bool get_cf_name_work(v8catalog* cf, OneC::Metadata::Configuration& config)
{
    if (!cf) return false;

    v8file* file = cf->GetFirst();
    while (file) {
        QString fileName = file->GetName();   // внутреннее имя в .cf

        // Разбираем заголовок объекта — только лёгкие поля
        QString name, synonym, type;
        QUuid   uuid;
        if (!parseObjectHeader(file, name, synonym, uuid, type)) {
            file = file->GetNext();
            continue;
        }

        // ✅ Формируем дескриптор и кладём его в нужный список Configuration
        OneC::Metadata::MetadataDescriptor desc(name, synonym, uuid, type, fileName);

        if (type == "Catalog")           config.addCatalog(desc);
        else if (type == "Document")     config.addDocument(desc);
        else if (type == "Report")       config.addReport(desc);
        else if (type == "DataProcessor")config.addDataProcessor(desc);
        else if (type == "CommonModule") config.addCommonModule(desc);
        // ... остальные типы ...

        file = file->GetNext();
    }

    return true;
}
